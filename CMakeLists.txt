cmake_minimum_required(VERSION 3.31)
project(KasaPro VERSION 4.3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Sql Network PrintSupport)
add_executable(KasaPro WIN32 main.cpp)
target_link_libraries(KasaPro PRIVATE Qt6::Widgets Qt6::Sql Qt6::Network Qt6::PrintSupport)
if(MSVC)
    target_compile_options(KasaPro PRIVATE /W3 /utf-8)
else()
    target_compile_options(KasaPro PRIVATE -Wall -Wextra)
endif()
